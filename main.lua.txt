--   script_key = "ALKA-FIRE-2024-HUB4"
--   loadstring(game:HttpGet("https://your-auth-bot.up.railway.app/api/script/alkahub"))()

-- Or host main.lua separately and point at the same bot URL:
--   script_key = "ALKA-FIRE-2024-HUB4"
--   loadstring(game:HttpGet("https://your-static-host/main.lua"))()

local BASE_URL = "https://alkaauthbot-production.up.railway.app"

-- ── Executor-safe HTTP ────────────────────────────────────────────
local function httpGet(url)
    if syn and syn.request then
        return syn.request({ Url = url, Method = "GET" }).Body
    elseif http and http.request then
        return http.request({ Url = url, Method = "GET" }).Body
    elseif request then
        return request({ Url = url, Method = "GET" }).Body
    else
        return game:HttpGet(url)
    end
end

-- ── Pull key from script_key global ──────────────────────────────
local inputKey = (typeof(script_key) == "string") and script_key:gsub("%s+", "") or nil

if not inputKey or inputKey == "" then
    warn("[AlkaHub] No key provided. Set script_key before loading.")
    return
end

-- ── Load and run KeySystem ────────────────────────────────────────
local keySource = httpGet(BASE_URL .. "/KeySystem.lua")
local KeySystem  = loadstring(keySource)()

KeySystem(inputKey, function(passed, reason)
    if not passed then
        -- reason is already a human-readable string from KeySystem.lua
        warn("[AlkaHub] " .. (reason or "Key rejected."))
        return
    end

    -- ── Everything below runs only after key passes ───────────────

    local function safeProtect(gui)
        if syn and syn.protect_gui then
            pcall(syn.protect_gui, gui)
            gui.Parent = game.CoreGui
        elseif gethui then
            gui.Parent = gethui()
        else
            gui.Parent = PlayerGui
        end
    end

    local function writeData(k, data)
        if writefile then
            pcall(writefile, "alkahub_" .. k .. ".json", HttpService:JSONEncode(data))
        end
    end

    local function readData(k)
        if readfile and isfile and isfile("alkahub_" .. k .. ".json") then
            local ok, res = pcall(readfile, "alkahub_" .. k .. ".json")
            if ok then
                local ok2, d = pcall(HttpService.JSONDecode, HttpService, res)
                if ok2 then return d end
            end
        end
        return nil
    end

    -- ── Hub state ─────────────────────────────────────────────────
    local Hub = { Version = "4.0", Toggles = {}, PanelPos = {} }
    local saved = readData("state")
    if saved then
        Hub.Toggles  = saved.Toggles  or {}
        Hub.PanelPos = saved.PanelPos or {}
    end

    local function saveState()
        writeData("state", { Toggles = Hub.Toggles, PanelPos = Hub.PanelPos })
    end

    local function mkFrame(parent, props)
        local f = Instance.new("Frame")
        for k, v in pairs(props) do f[k] = v end
        f.Parent = parent
        return f
    end

    local function mkLabel(parent, props)
        local l = Instance.new("TextLabel")
        l.BackgroundTransparency = 1
        l.TextColor3 = Color3.fromRGB(255, 255, 255)
        l.Font = Enum.Font.GothamBold
        for k, v in pairs(props) do l[k] = v end
        l.Parent = parent
        return l
    end

    local function mkButton(parent, props, cb)
        local b = Instance.new("TextButton")
        b.BackgroundColor3 = Color3.fromRGB(40, 40, 60)
        b.TextColor3 = Color3.fromRGB(255, 255, 255)
        b.Font = Enum.Font.GothamBold
        b.AutoButtonColor = false
        b.BorderSizePixel = 0
        for k, v in pairs(props) do b[k] = v end
        b.Parent = parent
        b.MouseButton1Click:Connect(function()
            TweenService:Create(b, TweenInfo.new(0.08), { BackgroundColor3 = Color3.fromRGB(80, 80, 120) }):Play()
            task.delay(0.12, function()
                TweenService:Create(b, TweenInfo.new(0.08), { BackgroundColor3 = Color3.fromRGB(40, 40, 60) }):Play()
            end)
            if cb then cb() end
        end)
        return b
    end

    local function mkToggle(parent, key, label, yPos, cb)
        local row = mkFrame(parent, {
            Size = UDim2.new(1, -20, 0, 32),
            Position = UDim2.new(0, 10, 0, yPos),
            BackgroundTransparency = 1,
        })
        mkLabel(row, {
            Size = UDim2.new(0.72, 0, 1, 0),
            Text = label,
            TextXAlignment = Enum.TextXAlignment.Left,
            TextSize = 13,
        })

        local track = mkFrame(row, {
            Size = UDim2.new(0, 44, 0, 24),
            Position = UDim2.new(1, -54, 0.5, -12),
            BackgroundColor3 = Color3.fromRGB(55, 55, 75),
        })
        Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)

        local knob = mkFrame(track, {
            Size = UDim2.new(0, 20, 0, 20),
            Position = UDim2.new(0, 2, 0.5, -10),
            BackgroundColor3 = Color3.fromRGB(190, 190, 210),
        })
        Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)

        local state = Hub.Toggles[key] or false

        local function refresh(on)
            TweenService:Create(knob, TweenInfo.new(0.15), {
                Position = on and UDim2.new(0, 22, 0.5, -10) or UDim2.new(0, 2, 0.5, -10)
            }):Play()
            TweenService:Create(track, TweenInfo.new(0.15), {
                BackgroundColor3 = on and Color3.fromRGB(100, 80, 220) or Color3.fromRGB(55, 55, 75)
            }):Play()
        end

        refresh(state)

        track.InputBegan:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch then
                state = not state
                Hub.Toggles[key] = state
                refresh(state)
                saveState()
                if cb then cb(state) end
            end
        end)

        return row
    end

    local function mkDraggable(panel, handle, key)
        local p = Hub.PanelPos[key]
        if p then panel.Position = UDim2.new(0, p.X, 0, p.Y) end

        local dragging, dragInput, dragStart, startPos = false, nil, nil, nil

        handle.InputBegan:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseButton1
            or i.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = i.Position
                startPos  = panel.Position
                i.Changed:Connect(function()
                    if i.UserInputState == Enum.UserInputState.End then dragging = false end
                end)
            end
        end)

        handle.InputChanged:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.MouseMovement
            or i.UserInputType == Enum.UserInputType.Touch then
                dragInput = i
            end
        end)

        UserInputService.InputChanged:Connect(function(i)
            if i == dragInput and dragging then
                local d = i.Position - dragStart
                local np = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                     startPos.Y.Scale, startPos.Y.Offset + d.Y)
                panel.Position = np
                Hub.PanelPos[key] = { X = np.X.Offset, Y = np.Y.Offset }
                saveState()
            end
        end)
    end
